our context free grammer rules

program ::= statemet* EOF

statement ::= definition ";" | expression ";" | "import" Identifier ";"

definition ::= Identifier "=" expression

expression ::= abstraction | application

abstraction ::= "lambda" Identifier "." expression | runtime_abstraction

runtime_abstraction ::= output_abstraction | input_abstraction

output_abstraction ::= "phi" "(" Integer ")" Identifier "." expression

input_abstraction ::= "psi" "(" Integer ")" Identifier "." expression

application ::= atom atom*

atom ::= Identifier | runtime_abstraction | "(" expression ")"

Identifier ::= IDENTIFIER ("->" IDENTIFIER)*

Integer ::= DIGIT+
